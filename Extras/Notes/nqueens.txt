#include <bits/stdc++.h>
using namespace std;

bool safe(vector<vector<int>>& board, int r, int c, int n) {
    for (int i = 0; i < r; i++) {
        if (board[i][c]) return false;
    }
    for (int i = r - 1, j = c - 1; i >= 0 && j >= 0; i--, j--) {
        if (board[i][j]) return false;
    }
    for (int i = r - 1, j = c + 1; i >= 0 && j < n; i--, j++) {
        if (board[i][j]) return false;
    }
    return true;
}

void solve(vector<vector<int>>& board, int r, int n) {
    if (r == n) {
        for (int i = 0; i < n; i++) {
            for (int j = 0; j < n; j++) {
                cout << (board[i][j] ? "Q " : ". ");
            }
            cout << "\n";
        }
        cout << "\n";
        return;
    }
    for (int c = 0; c < n; c++) {
        if (safe(board, r, c, n)) {
            board[r][c] = 1;
            solve(board, r + 1, n);
            board[r][c] = 0;
        }
    }
}

int main() {
    int n = 4;
    vector<vector<int>> board(n, vector<int>(n, 0));
    solve(board, 0, n);
}
