#include <bits/stdc++.h>
using namespace std;
vector<int> approxVertexCover(int n, vector<pair<int,int>> edges){
    vector<int> cover;
    vector<char> used(n,0);
    for(auto e: edges){
        int u=e.first, v=e.second;
        if(used[u]||used[v]) continue;
        used[u]=used[v]=1;
        cover.push_back(u); cover.push_back(v);
    }
    return cover;
}
int main(){
    int n=7;
    vector<pair<int,int>> edges = {{0,1},{0,2},{1,3},{2,3},{3,4},{4,5},{4,6}};
    vector<int> cover = approxVertexCover(n,edges);
    for(int v:cover) cout<<v<<" ";
    cout<<"\n";
}
