#include <bits/stdc++.h>
using namespace std;
void insertionSort(int a[], int n){
    a[n] = 1000;
    for(int i=0;i<n;i++){
        int key=a[i];
        int j = n+i;
        while (j>=1+i && a[j]>key){
            a[j+1] = a[j];
            j--;
        }
        a[j+1] = key;
    }
}
int main(){
    int a[100]={5,2,4,6,1,3};
    int n=6;
    insertionSort(a, n);
    for(int i=n; i<2*n; i++){
        cout << a[i] << " ";
    }
}
