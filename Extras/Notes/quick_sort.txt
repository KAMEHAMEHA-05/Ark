#include <bits/stdc++.h>
using namespace std;
int partition(vector<int>& a,int l,int r){
    int p=a[r],i=l;
    for(int j=l;j<r;j++) if(a[j]<p) swap(a[i++],a[j]);
    swap(a[i],a[r]);
    return i;
}
void quickSort(vector<int>& a,int l,int r){
    if(l<r){
        int pi=partition(a,l,r);
        quickSort(a,l,pi-1);
        quickSort(a,pi+1,r);
    }
}
int main(){
    vector<int> a={10,7,8,9,1,5};
    quickSort(a,0,a.size()-1);
    for(int x:a) cout<<x<<" ";
}
