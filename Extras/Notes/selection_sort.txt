#include <bits/stdc++.h>
using namespace std;
void selectionSort(vector<int>& a){
    for(int i=0;i<a.size()-1;i++){
        int m=i;
        for(int j=i+1;j<a.size();j++) if(a[j]<a[m]) m=j;
        swap(a[i],a[m]);
    }
}
int main(){
    vector<int> a={64,25,12,22,11};
    selectionSort(a);
    for(int x:a) cout<<x<<" ";
}
