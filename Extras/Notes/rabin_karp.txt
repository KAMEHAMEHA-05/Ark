#include <bits/stdc++.h>
using namespace std;

void rabinKarp(string t, string p, int d = 256, int q = 101) {
    int n = t.size();
    int m = p.size();
    int h = 1;
    int pHash = 0;
    int tHash = 0;

    for (int i = 0; i < m - 1; i++) {
        h = (h * d) % q;
    }

    for (int i = 0; i < m; i++) {
        pHash = (d * pHash + p[i]) % q;
        tHash = (d * tHash + t[i]) % q;
    }

    for (int i = 0; i <= n - m; i++) {
        if (pHash == tHash) {
            bool match = true;
            for (int j = 0; j < m; j++) {
                if (t[i + j] != p[j]) {
                    match = false;
                    break;
                }
            }
            if (match) {
                cout << "Pattern found at " << i << "\n";
            }
        }

        if (i < n - m) {
            tHash = (d * (tHash - t[i] * h) + t[i + m]) % q;
            if (tHash < 0) {
                tHash += q;
            }
        }
    }
}

int main() {
    rabinKarp("AABAACAADAABAABA", "AABA");
}
