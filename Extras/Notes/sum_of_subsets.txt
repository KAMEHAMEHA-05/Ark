#include <bits/stdc++.h>
using namespace std;

void subsetSum(vector<int>& a, int i, int sum, int target, vector<int>& cur) {
    if (sum == target) {
        for (int x : cur) {
            cout << x << " ";
        }
        cout << "\n";
        return;
    }

    if (i == a.size() || sum > target) return;

    cur.push_back(a[i]);
    subsetSum(a, i + 1, sum + a[i], target, cur);
    cur.pop_back();
    subsetSum(a, i + 1, sum, target, cur);
}

int main() {
    vector<int> a = {10,7,5,18,12,20,15};
    int target = 35;
    vector<int> cur;
    subsetSum(a, 0, 0, target, cur);
}
